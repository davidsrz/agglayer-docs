    Getting Started with Vault Bridge¶

    This guide provides step-by-step instructions for integrating Vault Bridge into your application or chain. Vault Bridge follows the ERC-4626 interface, making integration straightforward for developers familiar with vault protocols.

    Prerequisites¶

    Development Environment¶

    • Node.js: Version 16 or higher
    • Forge: Foundry development framework
    • Git: Version control system
    • Code Editor: VS Code, Vim, or your preferred editor

    Blockchain Knowledge¶

    • ERC-20 Tokens: Understanding of standard token operations
    • ERC-4626 Vaults: Familiarity with vault interfaces
    • Cross-Chain Bridges: Basic knowledge of bridge operations
    • Smart Contract Development: Solidity development experience

    Network Access¶

    • RPC Endpoints: Access to Ethereum and target Layer 2 networks
    • Test Tokens: Testnet tokens for development and testing
    • Bridge Access: Connection to Agglayer or supported bridge networks

    Installation¶

    Using npm¶

    npm install @agglayer/vault-bridge

    Using Forge¶

    forge install agglayer/vault-bridge

    Manual Installation¶

    git clone https://github.com/agglayer/vault-bridge.git
cd vault-bridge
forge install
npm install

    Quick Start¶

    1. Basic Integration¶

    The simplest way to integrate with Vault Bridge is using the standard ERC-4626 interface:

    // SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "@openzeppelin/contracts/interfaces/IERC4626.sol";
import "@openzeppelin/contracts/interfaces/IERC20.sol";

contract SimpleVaultBridgeIntegration {
    IERC4626 public immutable vaultBridgeToken;
    IERC20 public immutable underlyingAsset;

    constructor(address _vaultBridgeToken, address _underlyingAsset) {
        vaultBridgeToken = IERC4626(_vaultBridgeToken);
        underlyingAsset = IERC20(_underlyingAsset);
    }

    function depositToVault(uint256 amount) external {
        // Transfer underlying asset from user
        underlyingAsset.transferFrom(msg.sender, address(this), amount);

        // Approve Vault Bridge Token to spend the asset
        underlyingAsset.approve(address(vaultBridgeToken), amount);

        // Deposit and receive vbTokens
        uint256 shares = vaultBridgeToken.deposit(amount, msg.sender);

        emit Deposited(msg.sender, amount, shares);
    }

    function withdrawFromVault(uint256 shares) external {
        // Redeem vbTokens for underlying assets
        uint256 assets = vaultBridgeToken.redeem(shares, msg.sender, msg.sender);

        emit Withdrawn(msg.sender, shares, assets);
    }

    event Deposited(address indexed user, uint256 assets, uint256 shares);
    event Withdrawn(address indexed user, uint256 shares, uint256 assets);
}

    2. Cross-Chain Integration¶

    For applications requiring cross-chain functionality:

    // SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

interface IVaultBridgeToken {
    function depositAndBridge(
        uint256 assets,
        address receiver,
        uint32 destinationNetwork
    ) external returns (uint256 shares);

    function claimAndRedeem(
        uint256 shares,
        address receiver,
        address owner
    ) external returns (uint256 assets);
}

contract CrossChainVaultIntegration {
    IVaultBridgeToken public immutable vaultBridgeToken;
    uint32 public immutable destinationNetwork;

    constructor(address _vaultBridgeToken, uint32 _destinationNetwork) {
        vaultBridgeToken = IVaultBridgeToken(_vaultBridgeToken);
        destinationNetwork = _destinationNetwork;
    }

    function depositAndBridgeAssets(uint256 amount, address receiver) external {
        // This combines deposit and bridge operations in a single call
        uint256 shares = vaultBridgeToken.depositAndBridge(
            amount,
            receiver,
            destinationNetwork
        );

        emit DepositedAndBridged(msg.sender, receiver, amount, shares);
    }

    event DepositedAndBridged(
        address indexed depositor,
        address indexed receiver,
        uint256 amount,
        uint256 shares
    );
}

    Working with Different Asset Types¶

    WETH Integration¶

    contract WETHVaultIntegration {
    IERC4626 public immutable vbWETH;
    IWETH public immutable WETH;

    constructor(address _vbWETH, address _WETH) {
        vbWETH = IERC4626(_vbWETH);
        WETH = IWETH(_WETH);
    }

    function depositETH() external payable {
        // Convert ETH to WETH
        WETH.deposit{value: msg.value}();

        // Approve and deposit WETH to vault
        WETH.approve(address(vbWETH), msg.value);
        uint256 shares = vbWETH.deposit(msg.value, msg.sender);

        emit ETHDeposited(msg.sender, msg.value, shares);
    }

    function withdrawToETH(uint256 shares) external {
        // Redeem vbWETH for WETH
        uint256 wethAmount = vbWETH.redeem(shares, address(this), msg.sender);

        // Convert WETH back to ETH and send to user
        WETH.withdraw(wethAmount);
        payable(msg.sender).transfer(wethAmount);

        emit ETHWithdrawn(msg.sender, shares, wethAmount);
    }

    receive() external payable {}

    event ETHDeposited(address indexed user, uint256 ethAmount, uint256 shares);
    event ETHWithdrawn(address indexed user, uint256 shares, uint256 ethAmount);
}

    USDC Integration¶

    contract USDCVaultIntegration {
    IERC4626 public immutable vbUSDC;
    IERC20 public immutable USDC;
    uint256 public constant USDC_DECIMALS = 6;

    constructor(address _vbUSDC, address _USDC) {
        vbUSDC = IERC4626(_vbUSDC);
        USDC = IERC20(_USDC);
    }

    function depositUSDC(uint256 amount) external {
        require(amount > 0, "Amount must be greater than 0");

        // Transfer USDC from user
        USDC.transferFrom(msg.sender, address(this), amount);

        // Approve and deposit to vault
        USDC.approve(address(vbUSDC), amount);
        uint256 shares = vbUSDC.deposit(amount, msg.sender);

        emit USDCDeposited(msg.sender, amount, shares);
    }

    function getUSDCValue(uint256 shares) external view returns (uint256) {
        return vbUSDC.convertToAssets(shares);
    }

    event USDCDeposited(address indexed user, uint256 amount, uint256 shares);
}

    Testing Your Integration¶

    Local Testing Setup¶

    # Start local test environment
forge test --fork-url $ETHEREUM_RPC_URL

    Unit Test Example¶

    // test/VaultBridgeIntegration.t.sol
pragma solidity ^0.8.0;

import "forge-std/Test.sol";
import "../src/SimpleVaultBridgeIntegration.sol";

contract VaultBridgeIntegrationTest is Test {
    SimpleVaultBridgeIntegration integration;
    IERC20 mockUSDC;
    IERC4626 mockVbUSDC;

    address user = makeAddr("user");
    uint256 constant DEPOSIT_AMOUNT = 1000e6; // 1000 USDC

    function setUp() public {
        // Deploy mock contracts (implementation depends on your setup)
        mockUSDC = deployMockUSDC();
        mockVbUSDC = deployMockVbUSDC();

        // Deploy integration contract
        integration = new SimpleVaultBridgeIntegration(
            address(mockVbUSDC),
            address(mockUSDC)
        );

        // Setup user with USDC
        deal(address(mockUSDC), user, DEPOSIT_AMOUNT);
    }

    function testDeposit() public {
        vm.startPrank(user);

        // Approve integration to spend USDC
        mockUSDC.approve(address(integration), DEPOSIT_AMOUNT);

        // Perform deposit
        integration.depositToVault(DEPOSIT_AMOUNT);

        // Verify vbTokens received
        uint256 vbBalance = mockVbUSDC.balanceOf(user);
        assertGt(vbBalance, 0, "User should receive vbTokens");

        vm.stopPrank();
    }

    function testWithdraw() public {
        // First deposit
        vm.startPrank(user);
        mockUSDC.approve(address(integration), DEPOSIT_AMOUNT);
        integration.depositToVault(DEPOSIT_AMOUNT);

        uint256 shares = mockVbUSDC.balanceOf(user);

        // Then withdraw
        integration.withdrawFromVault(shares);

        // Verify USDC returned
        uint256 usdcBalance = mockUSDC.balanceOf(user);
        assertApproxEqAbs(usdcBalance, DEPOSIT_AMOUNT, 1, "Should return approximately same USDC");

        vm.stopPrank();
    }
}

    Working with Native Converters¶

    Layer 2 Integration¶

    For applications running on Layer 2 networks, you can use Native Converters to mint vbTokens directly:

    // SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

interface INativeConverter {
    function convert(uint256 amount, address receiver) external returns (uint256 vbTokens);
    function deconvert(uint256 vbTokenAmount, address receiver) external returns (uint256 assets);
    function bridgeAndDeconvert(uint256 vbTokenAmount, address receiver, uint32 destinationNetwork) external;
}

contract Layer2VaultIntegration {
    INativeConverter public immutable nativeConverter;
    IERC20 public immutable underlyingAsset;
    IERC20 public immutable vbToken;

    constructor(
        address _nativeConverter,
        address _underlyingAsset,
        address _vbToken
    ) {
        nativeConverter = INativeConverter(_nativeConverter);
        underlyingAsset = IERC20(_underlyingAsset);
        vbToken = IERC20(_vbToken);
    }

    function convertToVbToken(uint256 amount) external {
        // Transfer underlying asset from user
        underlyingAsset.transferFrom(msg.sender, address(this), amount);

        // Approve Native Converter
        underlyingAsset.approve(address(nativeConverter), amount);

        // Convert to vbTokens
        uint256 vbTokens = nativeConverter.convert(amount, msg.sender);

        emit Converted(msg.sender, amount, vbTokens);
    }

    function deconvertFromVbToken(uint256 vbTokenAmount) external {
        // Transfer vbTokens from user
        vbToken.transferFrom(msg.sender, address(this), vbTokenAmount);

        // Deconvert to underlying assets
        uint256 assets = nativeConverter.deconvert(vbTokenAmount, msg.sender);

        emit Deconverted(msg.sender, vbTokenAmount, assets);
    }

    event Converted(address indexed user, uint256 assets, uint256 vbTokens);
    event Deconverted(address indexed user, uint256 vbTokens, uint256 assets);
}

    Deployment Scripts¶

    Forge Deployment Script¶

    // script/DeployIntegration.s.sol
pragma solidity ^0.8.0;

import "forge-std/Script.sol";
import "../src/SimpleVaultBridgeIntegration.sol";

contract DeployIntegration is Script {
    function run() external {
        uint256 deployerPrivateKey = vm.envUint("PRIVATE_KEY");

        vm.startBroadcast(deployerPrivateKey);

        // Deploy on Ethereum mainnet
        if (block.chainid == 1) {
            deployEthereumContracts();
        }
        // Deploy on Polygon
        else if (block.chainid == 137) {
            deployPolygonContracts();
        }

        vm.stopBroadcast();
    }

    function deployEthereumContracts() internal {
        address vbWETH = 0x...; // Vault Bridge WETH address
        address WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;

        SimpleVaultBridgeIntegration integration = new SimpleVaultBridgeIntegration(
            vbWETH,
            WETH
        );

        console.log("Deployed SimpleVaultBridgeIntegration at:", address(integration));
    }

    function deployPolygonContracts() internal {
        address nativeConverter = 0x...; // Native Converter address
        address bridgedWETH = 0x...; // Bridged WETH address on Polygon
        address vbWETH = 0x...; // vbWETH address on Polygon

        Layer2VaultIntegration l2Integration = new Layer2VaultIntegration(
            nativeConverter,
            bridgedWETH,
            vbWETH
        );

        console.log("Deployed Layer2VaultIntegration at:", address(l2Integration));
    }
}

    Frontend Integration¶

    Web3 Integration Example¶

    // frontend/vaultBridge.js
import { ethers } from 'ethers';

class VaultBridgeIntegration {
    constructor(provider, vaultBridgeTokenAddress) {
        this.provider = provider;
        this.vaultBridgeToken = new ethers.Contract(
            vaultBridgeTokenAddress,
            VAULT_BRIDGE_ABI,
            provider
        );
    }

    async deposit(amount, userAddress) {
        const signer = this.provider.getSigner();
        const contract = this.vaultBridgeToken.connect(signer);

        try {
            const tx = await contract.deposit(amount, userAddress);
            await tx.wait();
            return tx.hash;
        } catch (error) {
            console.error('Deposit failed:', error);
            throw error;
        }
    }

    async withdraw(shares, userAddress) {
        const signer = this.provider.getSigner();
        const contract = this.vaultBridgeToken.connect(signer);

        try {
            const tx = await contract.redeem(shares, userAddress, userAddress);
            await tx.wait();
            return tx.hash;
        } catch (error) {
            console.error('Withdrawal failed:', error);
            throw error;
        }
    }

    async getBalance(userAddress) {
        return await this.vaultBridgeToken.balanceOf(userAddress);
    }

    async getAssetValue(shares) {
        return await this.vaultBridgeToken.convertToAssets(shares);
    }
}

// Usage example
const provider = new ethers.providers.Web3Provider(window.ethereum);
const vaultBridge = new VaultBridgeIntegration(
    provider,
    '0x...' // Vault Bridge Token address
);

// Deposit example
async function handleDeposit() {
    try {
        const amount = ethers.utils.parseEther('1.0'); // 1 ETH
        const userAddress = await provider.getSigner().getAddress();

        const txHash = await vaultBridge.deposit(amount, userAddress);
        console.log('Deposit successful:', txHash);
    } catch (error) {
        console.error('Deposit failed:', error);
    }
}

    Best Practices¶

    Security Considerations¶

    1. Input Validation: Always validate user inputs and amounts
    2. Access Control: Implement proper access controls for administrative functions
    3. Reentrancy Protection: Use OpenZeppelin’s ReentrancyGuard when needed
    4. Slippage Protection: Consider slippage when dealing with yield-bearing assets
    import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

contract SecureVaultIntegration is ReentrancyGuard {
    function depositWithSlippageProtection(
        uint256 amount,
        uint256 minShares
    ) external nonReentrant {
        uint256 shares = vaultBridgeToken.deposit(amount, msg.sender);
        require(shares >= minShares, "Slippage too high");
    }
}

    Gas Optimization¶

    1. Batch Operations: Combine multiple operations when possible
    2. Efficient Storage: Use packed structs and appropriate data types
    3. Event Optimization: Only emit necessary events with indexed parameters

    Error Handling¶

    contract RobustVaultIntegration {
    error InsufficientBalance(uint256 required, uint256 available);
    error InvalidAmount(uint256 amount);
    error DepositFailed(string reason);

    function safeDeposit(uint256 amount) external {
        if (amount == 0) revert InvalidAmount(amount);

        uint256 balance = underlyingAsset.balanceOf(msg.sender);
        if (balance < amount) revert InsufficientBalance(amount, balance);

        try vaultBridgeToken.deposit(amount, msg.sender) returns (uint256 shares) {
            emit DepositSuccessful(msg.sender, amount, shares);
        } catch Error(string memory reason) {
            revert DepositFailed(reason);
        }
    }
}
