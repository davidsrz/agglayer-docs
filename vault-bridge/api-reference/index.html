    Core Interfaces¶

    IVaultBridgeToken¶

    The main interface for Vault Bridge Token contracts, extending ERC-4626 with bridge functionality.

    interface IVaultBridgeToken is IERC4626 {
    // Bridge operations
    function depositAndBridge(
        uint256 assets,
        address receiver,
        uint32 destinationNetwork
    ) external returns (uint256 shares);

    function claimAndRedeem(
        uint256 shares,
        address receiver,
        address owner
    ) external returns (uint256 assets);

    // Migration support
    function receiveMigration(
        uint256 assets,
        address receiver
    ) external returns (uint256 shares);

    // Yield information
    function yieldVault() external view returns (address);
    function totalYield() external view returns (uint256);
    function yieldPerShare() external view returns (uint256);

    // Bridge configuration
    function bridgeExtension() external view returns (address);
    function migrationManager() external view returns (address);
}

    INativeConverter¶

    Interface for Native Converter contracts on secondary chains.

    interface INativeConverter is IERC20 {
    // Conversion operations
    function convert(
        uint256 assets,
        address receiver
    ) external returns (uint256 vbTokens);

    function deconvert(
        uint256 vbTokens,
        address receiver
    ) external returns (uint256 assets);

    // Bridge integration
    function convertAndBridge(
        uint256 assets,
        address receiver,
        uint32 destinationNetwork
    ) external returns (uint256 vbTokens);

    function bridgeAndDeconvert(
        uint256 vbTokens,
        address receiver,
        uint32 destinationNetwork
    ) external;

    // Liquidity management
    function availableLiquidity() external view returns (uint256);
    function totalLiquidity() external view returns (uint256);
    function migrationThreshold() external view returns (uint256);

    // Asset information
    function underlyingAsset() external view returns (address);
    function vbToken() external view returns (address);
    function conversionRate() external view returns (uint256);
}

    IMigrationManager¶

    Interface for Migration Manager contracts coordinating cross-chain asset migration.

    interface IMigrationManager {
    struct MigrationRequest {
        address vaultBridgeToken;
        uint256 assets;
        address receiver;
        uint32 sourceNetwork;
        bytes32 migrationId;
        bytes proof;
    }

    // Migration operations
    function processMigration(MigrationRequest calldata request) external;

    function batchMigrations(MigrationRequest[] calldata requests) external;

    function validateMigration(
        MigrationRequest calldata request
    ) external view returns (bool valid);

    // Configuration
    function addVaultBridgeToken(address vaultBridgeToken) external;
    function removeVaultBridgeToken(address vaultBridgeToken) external;

    // Query functions
    function supportedTokens() external view returns (address[] memory);
    function migrationStatus(bytes32 migrationId) external view returns (uint8);
    function pendingMigrations() external view returns (uint256);
}

    ICustomToken¶

    Interface for Custom Token implementations with enhanced functionality.

    interface ICustomToken is IERC20 {
    // Custom functionality
    function upgrade(uint256 vbTokenAmount) external returns (uint256 customTokens);
    function downgrade(uint256 customTokens) external returns (uint256 vbTokenAmount);

    // Integration support
    function baseToken() external view returns (address);
    function nativeConverter() external view returns (address);

    // Custom interface support
    function supportsInterface(bytes4 interfaceId) external view returns (bool);
}

    Functions¶

    Core ERC4626 Functions¶

    function deposit(uint256 assets, address receiver) 
    external 
    returns (uint256 shares);
    Parameters: - assets: Amount of underlying assets to deposit - receiver: Address to receive the vault shares

    Returns: - shares: Amount of vault shares minted

    Description: Deposits underlying assets and mints vault shares (vbTokens) in 1:1 ratio.

    function withdraw(
    uint256 assets,
    address receiver,
    address owner
) external returns (uint256 shares);
    Parameters: - assets: Amount of underlying assets to withdraw - receiver: Address to receive the assets - owner: Address that owns the shares being burned

    Returns: - shares: Amount of shares burned

    Description: Burns vault shares and withdraws underlying assets.

    function redeem(
    uint256 shares,
    address receiver,
    address owner
) external returns (uint256 assets);
    Parameters: - shares: Amount of vault shares to redeem - receiver: Address to receive the underlying assets - owner: Address that owns the shares

    Returns: - assets: Amount of underlying assets withdrawn

    Description: Redeems vault shares for underlying assets.

    Bridge Integration Functions¶

    function depositAndBridge(
    uint256 assets,
    address receiver,
    uint32 destinationNetwork
) external returns (uint256 shares);
    Parameters: - assets: Amount of underlying assets to deposit - receiver: Address on destination chain to receive vbTokens - destinationNetwork: Chain ID of destination network

    Returns: - shares: Amount of vault shares minted and bridged

    Description: Combines deposit and bridge operations in a single transaction.

    function claimAndRedeem(
    uint256 shares,
    address receiver,
    address owner
) external returns (uint256 assets);
    Parameters: - shares: Amount of vault shares to redeem - receiver: Address to receive underlying assets - owner: Address that owns the shares on source chain

    Returns: - assets: Amount of underlying assets withdrawn

    Description: Claims bridged vbTokens and immediately redeems them for underlying assets.

    Migration Functions¶

    function receiveMigration(
    uint256 assets,
    address receiver
) external onlyMigrationManager returns (uint256 shares);
    Parameters: - assets: Amount of underlying assets being migrated - receiver: Address to receive the newly minted vbTokens

    Returns: - shares: Amount of vault shares minted

    Description: Receives migrated assets from Layer 2 Native Converters and mints vbTokens.

    View Functions¶

    function totalAssets() external view returns (uint256);
    Returns: Total underlying assets controlled by the vault

    function convertToShares(uint256 assets) external view returns (uint256);
    Parameters: - assets: Amount of underlying assets

    Returns: Equivalent amount of vault shares

    function convertToAssets(uint256 shares) external view returns (uint256);
    Parameters: - shares: Amount of vault shares

    Returns: Equivalent amount of underlying assets

    function previewDeposit(uint256 assets) external view returns (uint256);
    Parameters: - assets: Amount of assets to deposit

    Returns: Amount of shares that would be minted

    function previewWithdraw(uint256 assets) external view returns (uint256);
    Parameters: - assets: Amount of assets to withdraw

    Returns: Amount of shares that would be burned

    function maxDeposit(address receiver) external view returns (uint256);
    Parameters: - receiver: Address that would receive the shares

    Returns: Maximum amount of assets that can be deposited

    function maxWithdraw(address owner) external view returns (uint256);
    Parameters: - owner: Address that owns the shares

    Returns: Maximum amount of assets that can be withdrawn

    NativeConverter API¶

    Functions¶

    function convert(uint256 assets, address receiver) 
    external 
    returns (uint256 vbTokens);
    Parameters: - assets: Amount of underlying assets to convert - receiver: Address to receive vbTokens

    Returns: - vbTokens: Amount of vbTokens minted

    Description: Converts bridged underlying assets to vbTokens on Layer 2.

    function deconvert(uint256 vbTokens, address receiver) 
    external 
    returns (uint256 assets);
    Parameters: - vbTokens: Amount of vbTokens to deconvert - receiver: Address to receive underlying assets

    Returns: - assets: Amount of underlying assets returned

    Description: Deconverts vbTokens back to underlying assets (subject to liquidity availability).

    function triggerMigration() external;
    Description: Manually triggers migration of accumulated backing to Layer 1 when threshold is reached.

    function availableLiquidity() external view returns (uint256);
    Returns: Amount of underlying assets available for immediate deconversion.

    MigrationManager API¶

    Functions¶

    function processMigration(MigrationRequest calldata request) external;
    Parameters: - request: Migration request containing all necessary data

    Description: Processes a single migration request from Layer 2 to Layer 1.

    function batchMigrations(MigrationRequest[] calldata requests) external;
    Parameters: - requests: Array of migration requests

    Description: Processes multiple migration requests in a single transaction for gas efficiency.

    function addVaultBridgeToken(address vaultBridgeToken) external onlyOwner;
    Parameters: - vaultBridgeToken: Address of Vault Bridge Token to support

    Description: Adds a new Vault Bridge Token to the supported tokens list.

    Integration Examples¶

    Basic Integration¶

    pragma solidity ^0.8.0;

import "./interfaces/IVaultBridgeToken.sol";

contract BasicIntegration {
    IVaultBridgeToken public vbToken;

    constructor(address _vbToken) {
        vbToken = IVaultBridgeToken(_vbToken);
    }

    function depositForUser(uint256 amount) external {
        // User must approve this contract first
        IERC20(vbToken.asset()).transferFrom(msg.sender, address(this), amount);
        IERC20(vbToken.asset()).approve(address(vbToken), amount);

        uint256 shares = vbToken.deposit(amount, msg.sender);

        emit UserDeposited(msg.sender, amount, shares);
    }

    event UserDeposited(address indexed user, uint256 amount, uint256 shares);
}

    Cross-Chain Integration¶

    pragma solidity ^0.8.0;

contract CrossChainIntegration {
    IVaultBridgeToken public vbToken;
    uint32 public destinationNetwork;

    constructor(address _vbToken, uint32 _destinationNetwork) {
        vbToken = IVaultBridgeToken(_vbToken);
        destinationNetwork = _destinationNetwork;
    }

    function depositAndBridgeForUser(uint256 amount, address recipient) external {
        IERC20(vbToken.asset()).transferFrom(msg.sender, address(this), amount);
        IERC20(vbToken.asset()).approve(address(vbToken), amount);

        uint256 shares = vbToken.depositAndBridge(
            amount,
            recipient,
            destinationNetwork
        );

        emit CrossChainDepositInitiated(
            msg.sender,
            recipient,
            amount,
            shares,
            destinationNetwork
        );
    }

    event CrossChainDepositInitiated(
        address indexed sender,
        address indexed recipient,
        uint256 amount,
        uint256 shares,
        uint32 destinationNetwork
    );
}

    Yield Farming Integration¶

    pragma solidity ^0.8.0;

contract YieldFarmIntegration {
    struct Farm {
        IVaultBridgeToken stakingToken;
        IERC20 rewardToken;
        uint256 rewardRate;
        uint256 lastUpdateTime;
        uint256 rewardPerTokenStored;
    }

    struct UserRewards {
        uint256 userRewardPerTokenPaid;
        uint256 rewards;
    }

    mapping(uint256 => Farm) public farms;
    mapping(uint256 => mapping(address => uint256)) public balances;
    mapping(uint256 => mapping(address => UserRewards)) public userRewards;

    function stake(uint256 farmId, uint256 amount) external {
        Farm storage farm = farms[farmId];

        updateReward(farmId, msg.sender);

        balances[farmId][msg.sender] += amount;
        farm.stakingToken.transferFrom(msg.sender, address(this), amount);

        emit Staked(msg.sender, farmId, amount);
    }

    function claimRewards(uint256 farmId) external {
        updateReward(farmId, msg.sender);

        uint256 reward = userRewards[farmId][msg.sender].rewards;
        if (reward > 0) {
            userRewards[farmId][msg.sender].rewards = 0;
            farms[farmId].rewardToken.transfer(msg.sender, reward);
            emit RewardClaimed(msg.sender, farmId, reward);
        }
    }

    function updateReward(uint256 farmId, address account) internal {
        Farm storage farm = farms[farmId];
        farm.rewardPerTokenStored = rewardPerToken(farmId);
        farm.lastUpdateTime = block.timestamp;

        if (account != address(0)) {
            userRewards[farmId][account].rewards = earned(farmId, account);
            userRewards[farmId][account].userRewardPerTokenPaid = farm.rewardPerTokenStored;
        }
    }

    function rewardPerToken(uint256 farmId) public view returns (uint256) {
        Farm storage farm = farms[farmId];
        uint256 totalStaked = farm.stakingToken.balanceOf(address(this));

        if (totalStaked == 0) {
            return farm.rewardPerTokenStored;
        }

        return farm.rewardPerTokenStored + 
            (((block.timestamp - farm.lastUpdateTime) * farm.rewardRate * 1e18) / totalStaked);
    }

    function earned(uint256 farmId, address account) public view returns (uint256) {
        return (balances[farmId][account] * 
            (rewardPerToken(farmId) - userRewards[farmId][account].userRewardPerTokenPaid)) / 1e18 +
            userRewards[farmId][account].rewards;
    }

    event Staked(address indexed user, uint256 indexed farmId, uint256 amount);
    event RewardClaimed(address indexed user, uint256 indexed farmId, uint256 amount);
}

    Gas Usage Estimates¶

    VaultBridgeToken Operations¶

    Operation Estimated Gas Notes
    deposit() ~150,000 Includes external vault deposit
    withdraw() ~120,000 Includes external vault withdrawal
    redeem() ~120,000 Similar to withdraw
    depositAndBridge() ~200,000 Includes bridge messaging
    claimAndRedeem() ~150,000 Cross-chain claim + redeem

    NativeConverter Operations¶

    Operation Estimated Gas Notes
    convert() ~80,000 Simple conversion
    deconvert() ~90,000 Includes liquidity check
    convertAndBridge() ~140,000 Conversion + bridge
    triggerMigration() ~180,000 Cross-chain migration

    MigrationManager Operations¶

    Operation Estimated Gas Notes
    processMigration() ~120,000 Single migration
    batchMigrations() ~80,000 + 40,000 per migration Batch efficiency

    Rate Limits and Constraints¶

    Default Limits¶

    • Maximum daily deposits per user: 1,000 operations
    • Maximum single deposit: 10,000 tokens (varies by asset)
    • Minimum deposit: 0.001 tokens (varies by asset)
    • Migration threshold: 1,000 tokens accumulated
    • Reserve ratio: 10% (1,000 basis points)

    Slippage Protections¶

    • Maximum slippage: 1% (100 basis points)
    • Price impact threshold: 0.5% (50 basis points)
    • Yield variance tolerance: 2% (200 basis points)
